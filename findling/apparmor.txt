#include <tunables/global>

profile findling flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Nur für den Start: Besitzer von /data setzen und zum Benutzer findling wechseln.
  capability chown,
  capability fowner,
  capability dac_override,
  capability setuid,
  capability setgid,
  file,
  signal (send) set=(kill,term,int,hup,cont),

  # S6 overlay
  /init ix,
  /bin/** ix,
  /sbin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /etc/s6-overlay/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,
  /run.sh rix,
  /spiegel.sh r,
  /tmp/** rwk,
  /data/** rw,

  # Das eigentliche Programm läuft in einem eigenen, engeren Profil.
  /usr/bin/python3.14 cx -> findling_app,

  profile findling_app flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    signal (receive) peer=*_findling,
    signal (receive) peer=unconfined,

    # Nur TCP und Namensauflösung. Kein Rohzugriff aufs Netz.
    network inet stream,
    network inet6 stream,
    network inet dgram,
    network inet6 dgram,
    network unix stream,

    # Programm und Bibliotheken, nur lesend.
    /usr/bin/python3.14 mr,
    /usr/lib/** mr,
    /lib/** mr,
    /opt/findling/ r,
    /opt/findling/** mr,
    /usr/share/zoneinfo/** r,
    /etc/ssl/** r,
    /etc/passwd r,
    /etc/group r,
    /etc/hosts r,
    /etc/resolv.conf r,
    /etc/nsswitch.conf r,
    /etc/localtime r,
    /proc/** r,
    /sys/devices/system/cpu/** r,
    /dev/null rw,
    /dev/urandom r,
    /dev/tty rw,
    /dev/console rw,
    /dev/pts/* rw,

    # Für die Ausgabe ins Protokoll.
    /run/ r,
    /run/** rw,

    # Nur diese Ordner dürfen beschrieben werden. Die Datenbank braucht auch gemeinsamen Speicher (m).
    /data/ rw,
    /data/** rwkm,
    /tmp/ rw,
    /tmp/** rwkm,

    # Speicherorte für Sicherungen (Netzlaufwerke aus Home Assistant), nur lesend.
    # Schreiben darf dort allein der Kopierdienst.
    /share/ r,
    /share/** r,

    # Für das eigene Zertifikat des Direktzugangs.
    /usr/bin/openssl rix,
  }
}
